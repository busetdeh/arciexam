function Dijkstra(G, w, s):
  dist[s] = 0
  for each v in V \ {s}:
    dist[v] = ∞, prev[v] = null
  Q = BuildPriorityQueue(V, dist)
  
  while !Q.isEmpty():
    u = Q.extractMin()
    for each (u, v) in Adj[u]:
      alt = dist[u] + w(u, v)
      if alt < dist[v]:
        dist[v] = alt
        prev[v] = u
        Q.decreaseKey(v, alt)
  return dist, prev
